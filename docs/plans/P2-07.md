**`AnswerComparator.cs` specification:**

Location: `Learning.Domain/Services/` — mirrors `SkillGraphValidator.cs` (pure static class in the Domain layer; no DI, no infrastructure dependencies). The class goes in `Domain` because it encodes domain-level correctness rules, matching the convention established by `SkillGraphValidator` and `QuizQuestionContentValidator` (the latter lives in `Application/Features/...` but the `AnswerComparator` is purer and belongs in Domain).

Signature:
```csharp
/// <summary>
/// Pure static helper for server-side answer correctness comparison.
/// One method per QuestionType — plain switch, no Strategy/Factory (CLAUDE.md rule 8).
/// Mirrors the shape of SkillGraphValidator (static, no DI).
/// </summary>
public static class AnswerComparator
{
    public static bool AreEqual(QuestionType type, string? studentPayload, string? correctAnswer)
    {
        // Guard: treat null/empty as incorrect (do not throw)
        if (string.IsNullOrEmpty(studentPayload) || string.IsNullOrEmpty(correctAnswer))
            return false;

        return type switch
        {
            QuestionType.MCQ =>
                string.Equals(studentPayload.Trim(), correctAnswer.Trim(), StringComparison.OrdinalIgnoreCase),

            QuestionType.TrueFalse =>
                bool.TryParse(studentPayload.Trim(), out var studentBool)
                && bool.TryParse(correctAnswer.Trim(), out var correctBool)
                && studentBool == correctBool,

            QuestionType.FillInBlank =>
                string.Equals(studentPayload.Trim(), correctAnswer.Trim(), StringComparison.OrdinalIgnoreCase),

            QuestionType.Matching =>
                // TODO P2-07.b: define Matching answer payload shape with FE;
                // current behavior falls through to string compare until the wire-shape is defined.
                string.Equals(studentPayload.Trim(), correctAnswer.Trim(), StringComparison.OrdinalIgnoreCase),

            _ => false,
        };
    }
}
```

// P2-07: Publish AnswerSubmittedIntegrationEvent (Option B — direct IPublisher, fail-soft).
// Fires inside the UoW transaction (before CommitAsync). A commit failure after a successful
// publish leaks a ghost event — accepted trade-off for Phase-2 MVP per ADR 0002 §3.
// A publisher failure (e.g. downstream handler throws) does NOT fail the user request
// because IsolatedNotificationPublisher catches and logs per-handler exceptions, and the

if (question.SkillId.HasValue)

    try
    {
        await _publisher.Publish(new AnswerSubmittedIntegrationEvent(
            EventId: Guid.NewGuid(),
            OccurredOnUtc: DateTime.UtcNow,
            StudentId: studentId.Value,
            LessonId: attempt.LessonId,
            SkillId: question.SkillId.Value,
            CorrectAnswerCount: isCorrect ? 1 : 0),
            cancellationToken);
    }
    catch (Exception ex)
    {
        // Fail-soft: log but do NOT fail the student's submission request.
        // The answer is already staged; the event emission is best-effort.
        _logger.LogError(ex, "Failed to publish AnswerSubmittedIntegrationEvent for question {QuestionId}", question.Id);
    }
}
else
{
    _logger.LogWarning("AnswerSubmittedIntegrationEvent skipped — question {QuestionId} has no SkillId", question.Id);
