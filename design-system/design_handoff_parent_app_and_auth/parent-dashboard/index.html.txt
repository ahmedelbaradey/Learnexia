function App() {
  const [page, setPage] = React.useState('splash');
  const [addChildOpen, setAddChildOpen] = React.useState(false);
  const [child, setChild] = React.useState(PD_CHILDREN[0]);
  const [role, setRole] = React.useState('parent');
  const [theme, setTheme] = React.useState('night');

  React.useEffect(() => {
    document.documentElement.setAttribute('data-theme', theme);
  }, [theme]);

  // URL for the fake browser chrome
  const urlFor = (p) => ({
    splash:   'learnexia.com',
    role:     'learnexia.com/welcome',
    landing:  'learnexia.com',
    login:    'app.learnexia.com/login',
    register: 'app.learnexia.com/register',
    children: 'app.learnexia.com/children',
    overview: 'app.learnexia.com/sami/overview',
    reports:  'app.learnexia.com/sami/reports',
    settings: 'app.learnexia.com/settings',
  }[p] || 'app.learnexia.com');

  const sidebarFor = (p) => ({
    children: 'children', overview: 'overview', reports: 'reports', energy: 'energy', settings: 'settings',
  }[p] || 'overview');

  const onNav = (id) => {
    if (id === 'children') setPage('children');
    else if (id === 'overview') setPage('overview');
    else if (id === 'reports')  setPage('reports');
    else if (id === 'energy')   setPage('energy');
    else if (id === 'settings') setPage('settings');
    else setPage('overview'); // activity, subjects → fall back
  };

  const pages = {
    splash:   <SplashWebPage onContinue={() => setPage('role')}/>,
    role:     <RoleSelectWebPage onPick={(r) => { setRole(r); setPage('login'); }}/>,
    landing:  <LandingPage onLogin={() => setPage('role')} onSignup={() => setPage('register')}/>,
    login:    <LoginWebPage role={role} onLogin={() => setPage('children')} onRegister={() => setPage('register')} onLanding={() => setPage('landing')} onBack={() => setPage('role')}/>,
    register: <RegisterWebPage onRegister={() => setPage('children')} onLogin={() => setPage('login')} onLanding={() => setPage('landing')}/>,
    children: <MyChildrenWebPage sidebarActive={sidebarFor('children')} onNav={onNav}
                onPick={() => setPage('overview')} onAddChild={() => setAddChildOpen(true)}/>,
    overview: <OverviewWebPage  sidebarActive={sidebarFor('overview')} onNav={onNav}/>,
    reports:  <ReportsWebPage   sidebarActive={sidebarFor('reports')}  onNav={onNav}/>,
    energy:   <EnergyWebPage    sidebarActive={sidebarFor('energy')}   onNav={onNav}/>,
    settings: <SettingsWebPage  sidebarActive={sidebarFor('settings')} onNav={onNav}/>,
  };

  const groups = [
    { label: 'Public', items: [['splash','Splash'], ['role','Role Select'], ['landing','Landing'], ['login','Login'], ['register','Register']] },
    { label: 'App',    items: [['children','My Children'], ['overview','Dashboard'], ['reports','Reports'], ['energy','⚡ Energy'], ['settings','Settings']] },
  ];

  return (
    <PDCtx.Provider value={{ child, setChild, theme, setTheme, onLogout: () => setPage('login') }}>
    <div data-screen-label={`Web · ${page}`}>
      <div className="topbar">
        {groups.map((g, gi) => (
          <React.Fragment key={g.label}>
            {gi > 0 && <div className="divider"/>}
            <span className="label">{g.label}</span>
            {g.items.map(([id, label]) => (
              <button key={id} className={`crumb-w${page === id ? ' active' : ''}`} onClick={() => setPage(id)}>{label}</button>
            ))}
          </React.Fragment>
        ))}
      </div>
      <div className="frame">
        <div className="chrome">
          <div className="dots">
            <div className="dot" style={{ background: '#FF5F57' }}/>
            <div className="dot" style={{ background: '#FEBC2E' }}/>
            <div className="dot" style={{ background: '#28C840' }}/>
          </div>
          <div className="url">🔒 {urlFor(page)}</div>
          <div style={{ color: '#64748B', fontSize: 14 }}>⋯</div>
        </div>
        {pages[page]}
      </div>
      <AddChildModal open={addChildOpen} onClose={() => setAddChildOpen(false)} ar={false}/>
    </div>
    </PDCtx.Provider>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<App/>);
</script>
